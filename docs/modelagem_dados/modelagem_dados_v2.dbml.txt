// ============================================================
// Sistema de Gestão de Estágios — Fonoaudiologia UFCSPA
// Modelagem relacional — VERSÃO 2 (agnóstica de SGBD)
// Fiel à proposta de orquestração v2 e ao protótipo:
// ciclo como máquina de estados, motor de alocação (alocacoes +
// sessoes), grupos (grade-primeiro), preceptores/cobertura,
// afastamentos, indisponibilidades, eventos, remanejo, restrições
// de local do aluno, histórico (snapshot) e autenticação.
// ============================================================

// FONTE DE VERDADE: o schema real é dono do Alembic (app/models/ +
//   alembic/versions/). Este DBML é o companion visual (dbdiagram) e o
//   .sql é o companion DDL — mantidos em sincronia com os models.
//   Última migration refletida aqui: 9fd4120b1555 (AR-7 docente nullable +
//   AR-8 aluno.prioridade). Regras de motor/tela: docs/REGRAS_MOTOR_ESCALA.md.

// ============================================================
// REGRAS DE NEGÓCIO (v2 — fase/pré-requisito/carry-forward)
//  1. Fase do ciclo: alunos do 7º semestre formam um MINI-CICLO que cursa
//     SOMENTE Audiologia I (fase '7'). Alunos do 9º/10º cursam as demais
//     áreas (fase '9_10'). Ambos convivem no mesmo ciclo (ciclos.id).
//  2. Audiologia I é PRÉ-REQUISITO (areas.pre_requisito) das áreas de 9/10:
//     responsabilidade compartilhada, conferida pela coordenação.
//  3. Carry-forward: aluno de 9/10 começa com Audiologia I concluída, e pode
//     iniciar o ciclo com outras áreas já concluídas (cursadas antes). Isso é
//     uma matrícula nascendo com status='concluida'; o motor só aloca as
//     'em_andamento', então as concluídas são puladas naturalmente. Uma
//     matrícula pode fechar o período com feitos < N -> status='incompleta'
//     (carry-forward para o próximo ciclo; grade-primeiro §10.5).
//  4. Carga por área na matriz (matriz curricular). As áreas COMPOSTAS
//     (Audiologia II, Hospitalar) são containers: a CH da mãe = soma das
//     sub-áreas; matrícula e conclusão são POR SUB-ÁREA (leaf).
//  5. RESPONSÁVEIS DO LOCAL / COBERTURA: cada local tem um docente (UFCSPA) e,
//     opcionalmente, um PRECEPTOR de campo. O preceptor pode ser externo
//     (catálogo preceptores) OU um docente (referência polimórfica
//     locais.preceptor_tipo/preceptor_id). Para um encontro acontecer numa data
//     basta que UM dos responsáveis esteja disponível; o encontro só cai quando
//     docente E preceptor estiverem afastados ao mesmo tempo. Preceptores são
//     externos à UFCSPA, sem login institucional — a conta é o e-mail. Docentes,
//     preceptores e alunos têm e-mail para as notificações do sistema.
//  6. RESTRIÇÕES DE LOCAL DO ALUNO: por padrão o aluno pode ir a todos os locais;
//     `restricoes_aluno_local` guarda só as exceções (locais bloqueados por
//     condição especial). O motor pula esses locais ao alocar o aluno — que
//     segue cursando a área em outro local liberado. Não pode sobrar área
//     matriculada em_andamento sem nenhum local liberado (validação de aplicação).
//  7. GRUPOS (ondas por local — grade-primeiro): a capacidade do local = tamanho
//     do grupo. O molde de TODOS os grupos do ciclo é MATERIALIZADO no bootstrap
//     (datas = infraestrutura: dia fixo, pulando feriado/sem-cobertura, em blocos
//     de N = teto(carga / horas_sessao)) e é a FONTE DE VERDADE. onda 1 = grupo
//     atual; 2,3… = previstos (cascata: cada onda começa quando a anterior do
//     local conclui; com passagem_grupo, no último dia da anterior). Membro
//     grupo_alunos.fixado sobrevive à regeração (remanejo manual da coordenação).
//  8. MESMO DIA: sem sobreposição de horário + intervalo mínimo de 1h30 entre
//     áreas no mesmo dia (verificado com dia_semana/hora_inicio/hora_fim; áreas
//     diferentes PODEM dividir o dia, inclusive o mesmo turno). Teto de 30h/semana.
// ============================================================

// ---------- ENUMS ----------
Enum turno_tipo {
  "manha"
  "tarde"
  "integral"   // cenário de dia inteiro (ex.: Audiologia II - SM, EESCA NEB)
  "noite"
}

Enum fase_area {
  "7"      // 7º semestre (mini-ciclo): só Audiologia I
  "9_10"   // 9º/10º semestre: demais estágios
}

Enum dia_semana_tipo {
  "segunda"
  "terca"
  "quarta"
  "quinta"
  "sexta"
  "sabado"
  "domingo"
}

Enum status_ciclo {
  "rascunho"       // aberto, em bootstrap (retomável pelo passo_bootstrap)
  "em_andamento"   // escala confirmada, operação corrente
  "encerrado"      // fechado; gerou histórico
}

Enum status_matricula {
  "em_andamento"
  "concluida"
  "interrompida"   // aluno parou por motivo extraordinário (§6.1)
  "incompleta"     // período da caixa fechou com feitos < N -> carry-forward (§10.5)
}

Enum status_alocacao {
  "ativa"
  "concluida"
  "cancelada"
}

Enum status_sessao {
  "prevista"
  "cumprida"
  "remanejada"
  "cancelada"
}

Enum status_grupo {
  "em_andamento"   // onda atual (alunos hoje alocados no local)
  "previsto"       // onda futura projetada a partir da fila
}

Enum tipo_afastamento {
  "ferias"

  "outro"

Enum tipo_evento {

  "outro"

Enum origem_evento {

  "ciclo_completo"
  "pendente"
}

Enum perfil_usuario {
  "administrador"

  "docente"
  "aluno"

// ============================================================
// CICLO  (agregador + máquina de estados — o estado decide a tela)
// ============================================================

Table ciclos [headercolor: #16a085] {
  id                    int            [pk, increment]
  data_inicio           date           [not null]
  data_fim              date           [not null, note: 'Validação: data_fim > data_inicio']
  status                status_ciclo   [not null, default: 'rascunho']
  passo_bootstrap       int            [note: 'Passo do carrossel onde o rascunho parou (nulo fora do rascunho)']
  escala_desatualizada  boolean        [not null, default: false, note: 'Liga o alerta de Remanejar no painel']
  criado_em             timestamp
  encerrado_em          timestamp

  Note: 'Espinha dorsal da orquestração. Só um ciclo rascunho OU em_andamento por vez (invariante de aplicação). rascunho -> em_andamento ao confirmar o bootstrap; em_andamento -> encerrado no fim do ano.'
}

// ============================================================
// CATÁLOGOS PERMANENTES  (atravessam ciclos — fora do assistente)
// ============================================================

Table areas [headercolor: #3498db] {
  id              int          [pk, increment]
  nome            varchar      [not null, unique, note: 'Ex.: Audiologia I, Motricidade Orofacial, LAD (Linguagem do Adulto/Idoso)']
  cor             varchar(7)   [note: 'Cor (#hex) da pílula/badge da área na UI']
  carga_exigida   int          [not null, note: 'Horas obrigatórias para concluir a área/sub-área (matriz curricular). Validação: > 0']
  fase            fase_area    [not null, default: '9_10', note: 'Fase do curso: 7 (mini-ciclo, só Audiologia I) ou 9_10 (demais)']
  pre_requisito   boolean      [not null, default: false, note: 'True só em Audiologia I: responsabilidade compartilhada p/ as áreas de 9/10. Índice único parcial: no máximo UMA área pré-requisito.']
  composta        boolean      [not null, default: false, note: 'Área-mãe/container (ex.: Audiologia II, Hospitalar): NÃO é matriculável nem alocável (sem locais). O aluno cursa as SUB-ÁREAS (leaf).']
  area_mae_id     int          [ref: > areas.id, note: 'Sub-área: FK self para a área-mãe composta. Área simples ou mãe = NULL. A CH da mãe = soma das CHs das sub-áreas; matrícula/conclusão são por sub-área.']

  Note: 'Catálogo fixo das áreas do curso. Independe de ciclo — já vem pré-cadastrado. Áreas compostas agregam sub-áreas (Audiologia II, Hospitalar).'
}

Table docentes [headercolor: #3498db] {
  id     int       [pk, increment]
  nome   varchar   [not null, unique]
  email  varchar   [note: 'Login institucional UFCSPA + destino das notificações do sistema']
  ativo  boolean   [not null, default: true, note: 'Desligar sem apagar histórico; desligar em andamento dispara remanejo']

  Note: 'Professores da UFCSPA. Sobrevive ao ciclo (o mesmo professor atua em vários anos). O vínculo com o ciclo surge via locais e afastamentos — docente nunca guarda datas de ciclo. Um docente também pode ser o PRECEPTOR de campo de um local (ver locais.preceptor_tipo=docente).'
}

Table preceptores [headercolor: #3498db] {
  id     int       [pk, increment]
  nome   varchar   [not null]
  email  varchar   [not null, note: 'A conta do preceptor: são externos à UFCSPA e NÃO têm login institucional — a conta é gerada a partir do e-mail. Também é o destino das notificações.']
  ativo  boolean   [not null, default: true, note: 'Desligar em andamento dispara remanejo nos locais em que é responsável']

  Note: 'Fonoaudiólogas(os) de campo — responsáveis pelo local, externos à UFCSPA. Catálogo permanente (atravessa ciclos). Um preceptor pode responder por N locais (1 preceptor ↔ N locais). O responsável de campo de um local pode ser um preceptor (aqui) OU um docente (referência polimórfica em locais).'
}

// ============================================================
// ALUNOS  (por ciclo) + matrículas por área
// ============================================================

Table alunos [headercolor: #27ae60] {
  id            int        [pk, increment]
  ciclo_id      int        [not null, ref: > ciclos.id, note: 'O aluno pertence a um ciclo — permite arquivar por ano sem misturar turmas']
  nome          varchar    [not null]
  matricula     varchar    [not null]
  email         varchar    [note: 'Destino das notificações do sistema (padrão matricula@aluno.ufcspa.edu.br)']
  semestre      int        [note: 'Semestre atual. 7 => fase 7 (mini-ciclo Audiologia I); 9/10 => fase 9_10 (demais)']
  ordenamento   int        [not null, note: 'Ordem de alocação — menor = maior prioridade. DERIVADA (AR-8): por fase, prioritários primeiro, depois por matrícula — não é mais ordenação manual']
  prioridade    boolean    [not null, default: false, note: 'AR-8: marcado no bootstrap p/ POSICIONAR o aluno à mão na Montagem dos grupos (o motor honra a colocação/pin)']
  criado_em     timestamp

  Indexes {
    ordenamento             [name: 'idx_alunos_ordenamento']
    (ciclo_id, matricula)   [unique, name: 'uq_aluno_ciclo_matricula']
  }
}

// Junção aluno × área (substitui areasMatriculadas/concluidas e os
// campos dinâmicos data_matricula_<area> do modelo Mongo da v1)
Table matriculas [headercolor: #27ae60] {
  id                        int                [pk, increment]
  aluno_id                  int                [not null, ref: > alunos.id]
  area_id                   int                [not null, ref: > areas.id]
  data_matricula            date
  status                    status_matricula   [not null, default: 'em_andamento']
  data_conclusao_prevista   date               [note: 'Calculada pelo motor: data da última sessão que fecha a carga']
  data_conclusao            date               [note: 'Preenchida quando dia atual >= previsão e carga fechada']
  motivo_interrupcao        varchar            [note: 'Interrupção do estágio (desmatrícula por motivo extraordinário) — §6.1']
  data_interrupcao          date               [note: 'Data da interrupção (status=interrompida)']

  Indexes {
    (aluno_id, area_id) [unique, name: 'uq_matricula_aluno_area']
  }
  Note: 'Diz O QUE o aluno cursa (por sub-área quando a área é composta). Área do catálogo sem matrícula = estado "a iniciar" (derivado). Conclusão é automática (por encontros feitos), nunca check manual. CARRY-FORWARD: pode nascer com status=concluida (área já cursada antes) ou fechar o período com incompleta. PRÉ-REQUISITO: Audiologia I é responsabilidade compartilhada das áreas de 9/10.'
}

// Restrições de local do aluno (BLOCKLIST): por padrão o aluno está disponível
// em TODOS os locais; cada linha aqui é uma EXCEÇÃO (local que ele NÃO pode
// frequentar, por condição especial). O motor pula esses locais para o aluno.
Table restricoes_aluno_local [headercolor: #27ae60] {
  id        int   [pk, increment]
  aluno_id  int   [not null, ref: > alunos.id]
  local_id  int   [not null, ref: > locais.id]

  Indexes {
    (aluno_id, local_id) [unique, name: 'uq_restricao_aluno_local']
  }
  Note: 'Blocklist aluno × local. Ausência de linha = disponível (padrão). Granularidade por LOCAL — o aluno segue cursando a área em outro local liberado. Validação (aplicação): não deixar uma área matriculada em_andamento sem nenhum local liberado. FKs com ON DELETE CASCADE.'
}

// ============================================================
// LOCAIS DE ESTÁGIO  (SLOT: 1 local = 1 campo + dia + turno)
// ============================================================

Table locais [headercolor: #e67e22] {
  id             int               [pk, increment]
  ciclo_id       int               [not null, ref: > ciclos.id, note: 'A oferta muda de ano para ano']
  area_id        int               [not null, ref: > areas.id]
  unidade        varchar           [note: 'Unidade concedente do espelho: SMS, Santa Casa (HSCL/HCSA), GHC, UFCSPA, Frei Pacífico, Rádio Guaíba…']
  campo          varchar           [not null, note: 'Cenário físico (texto): AMB. IAPI, UTI Neonatal, Coral, Clínica…']
  docente_id     int               [ref: > docentes.id, note: 'Docente responsável (UFCSPA). AR-7: NULLABLE — o slot pode nascer sem docente e recebê-lo em Configurações de campo (migration 9fd4120b1555). "Todo local ativo tem docente" vira validação antes de gerar a escala.']
  preceptor_tipo varchar           [note: "Tipo do preceptor de campo: 'externo' (ref preceptores) | 'docente' (ref docentes) | null (sem preceptor separado, só o docente responde)"]
  preceptor_id   int               [note: 'Preceptor de campo (opcional). FK POLIMÓRFICA resolvida por preceptor_tipo: preceptores.id quando externo, docentes.id quando docente. CHECK ck_local_preceptor: ambos nulos OU (tipo válido E id não-nulo).']
  dia_semana     dia_semana_tipo   [not null, note: 'Modelo SLOT: 1 local = 1 (campo+dia+turno). Um campo pode ter N slots (dias/turnos distintos, inclusive o mesmo dia em turnos diferentes) que rodam em paralelo.']
  turno          turno_tipo        [not null]
  hora_inicio       time           [not null, note: 'Usados na sobreposição de horário e no intervalo mínimo de 1h30 entre áreas no mesmo dia']
  hora_fim          time           [not null, note: 'Validação: hora_fim > hora_inicio']
  capacidade        int            [not null, note: 'Vagas simultâneas = tamanho do grupo (antigo grupo_aluno). Validação: > 0']
  carga_horaria     int            [not null, note: 'Horas que o estágio nesse local cobre (consequência dos encontros × duração)']
  horas_sessao      float          [note: 'Horas REAIS de cada encontro (desconta almoço no integral). Base do numero_encontros e do teto de 30h/semana.']
  numero_encontros  int            [not null, note: 'Nº de encontros do cenário = teto(carga_horaria / horas_sessao). TOTAL fixo do contador; o motor agenda exatamente esse nº de sessões.']
  passagem_grupo    boolean        [not null, default: false, note: 'PASSAGEM DE GRUPO: se true, o último encontro de um grupo é o primeiro do próximo (1 dia de sobreposição). Na cascata de grupos, a onda seguinte começa no último dia da anterior; se false, começa depois.']
  ativo             boolean        [not null, default: true]

  Note: 'Cenário de prática (SLOT) de um ciclo. Desativar em andamento dispara remanejo. COBERTURA: para um encontro acontecer numa data basta que UM dos responsáveis do local (docente OU preceptor de campo) esteja disponível; o encontro só cai quando docente E preceptor estiverem afastados ao mesmo tempo. Local sem preceptor (preceptor_id null) depende só do docente.'
}

Table indisponibilidades_local [headercolor: #9b59b6] {

  motivo       varchar
  data_inicio  date     [not null]
  data_fim     date     [not null, note: 'Validação: data_fim >= data_inicio']

  Note: 'Só existe na Operação (não no bootstrap). Ao fim do período o local volta a valer sozinho.'

// ============================================================
// SAÍDA DO MOTOR DE ALOCAÇÃO
//   matriculas = O QUE  ·  alocacoes = ONDE  ·  sessoes = QUANDO
// ============================================================

Table alocacoes [headercolor: #e67e22] {
  id                  int               [pk, increment]
  aluno_id            int               [not null, ref: > alunos.id]
  local_id            int               [not null, ref: > locais.id, note: 'Área/campo/docente derivam do local — não duplicar']
  matricula_id        int               [not null, ref: > matriculas.id, note: 'Amarra a alocação à matrícula que a originou']
  data_inicio         date              [not null]
  data_fim_prevista   date              [not null, note: 'Recalculada a cada remanejamento']
  travada             boolean           [not null, default: false, note: 'Trava manual: o motor nunca mexe numa alocação travada']
  ajuste_encontros    int               [not null, default: 0, note: 'Ajuste manual do professor no contador de encontros: +1 reforço, -1 falta']
  status              status_alocacao   [not null, default: 'ativa']

  Indexes {
    (aluno_id, local_id) [unique, name: 'uq_alocacao_aluno_local']
  }
  Note: 'Aluno ocupando um local. Recriado na geração; ajustado cirurgicamente no remanejo.'
}

// As datas concretas — a menor unidade da escala (era o array calendario[])
Table sessoes [headercolor: #e67e22] {
  id             int             [pk, increment]
  alocacao_id    int             [not null, ref: > alocacoes.id]
  data           date            [not null]
  hora_inicio    time            [note: 'Herdada do local, mas gravada na sessão (histórico não muda se o local mudar)']
  hora_fim       time
  horas          decimal(6,2)
  status         status_sessao   [not null, default: 'prevista']

  Indexes {
    (alocacao_id, data) [name: 'idx_sessao_alocacao_data']
  }
  Note: 'A menor unidade da escala = o ENCONTRO. O motor gera exatamente locais.numero_encontros sessões. Encontro passado e prevista -> cumprida. O contador feitos/total é derivado: total = locais.numero_encontros; feitos = cumpridas + alocacoes.ajuste_encontros (limitado a [0,total]).'
}

// ============================================================
// GRUPOS  (ondas por local — molde grade-primeiro materializado no bootstrap)
// ============================================================

Table grupos [headercolor: #e67e22] {
  id            int           [pk, increment]
  ciclo_id      int           [not null, ref: > ciclos.id]
  local_id      int           [not null, ref: > locais.id]
  area_id       int           [not null, ref: > areas.id, note: 'Derivável do local; guardado por conveniência da projeção']
  onda          int           [not null, note: '1 = grupo atual; 2,3… = ondas seguintes (cascata)']
  status        status_grupo  [not null]
  data_inicio   date          [not null, note: 'Onda prevista: começa quando a onda anterior do local conclui']
  data_fim      date          [not null, note: 'Início + duração dos encontros (previsão)']

  Note: 'Uma onda de alunos ocupando um local numa janela ("caixa"). Capacidade do local = tamanho do grupo. GRADE-PRIMEIRO: o molde de TODOS os grupos do ciclo é materializado no bootstrap (datas = infraestrutura) e é a FONTE DE VERDADE; a onda em andamento tem datas comprometidas, as previstas re-derivam. FKs ciclo/local com ON DELETE CASCADE.'
}

Table grupo_alunos [headercolor: #e67e22] {

  Indexes {
    (grupo_id, aluno_id) [unique, name: 'uq_grupo_aluno']
  }
  Note: 'Membro de um grupo. FKs grupo/aluno com ON DELETE CASCADE.'
}

// OPERAÇÃO  (fila de remanejo + feed de atividade)
// ============================================================

Table fila_remanejo [headercolor: #c0392b] {
  id        int       [pk, increment]
  ciclo_id  int       [not null, ref: > ciclos.id]
  quando    date      [not null]
  texto     varchar   [not null, note: 'Gatilho acumulado (docente afastado, local indisponível, evento importado…)']

  Note: 'Gatilhos que mantêm escala_desatualizada = true até o Remanejamento ser aplicado.'
}

Table atividade [headercolor: #c0392b] {
  id        int              [pk, increment]
  ciclo_id  int              [not null, ref: > ciclos.id]
  quando    date             [not null]
  texto     varchar          [not null]
  tipo      tipo_atividade   [not null, default: 'edicao']

  Note: 'Feed de atividade recente do Painel de Operação.'
}

// ============================================================
// HISTÓRICO  (snapshot denormalizado de propósito — o passado não muda)
// ============================================================

Table historico [headercolor: #34495e] {
  id                   int                  [pk, increment]
  ciclo_id             int                  [not null, ref: > ciclos.id, note: 'Vínculo para auditoria/drill-down']
  ano                  int                  [not null, note: 'Chave de navegação da tela de Histórico (2026 · 2025…)']
  aluno_nome           varchar              [not null, note: 'Cópia (snapshot), não FK — o aluno pode ser editado/removido depois']
  matricula            varchar              [note: 'Cópia (snapshot)']
  areas                json                 [note: 'Array por área/sub-área: {nome, carga_exigida, horas_cumpridas, data_conclusao} — inclui as não concluídas. Persistido como JSONB.']
  carga_horaria_total  int                  [note: 'Soma das horas cumpridas no ciclo']
  situacao             situacao_historico   [not null, note: 'ciclo_completo (todas as áreas da fase concluídas) | pendente']
  encerramento         date
  criado_em            timestamp

  Note: 'Retrato gravado no encerramento. Denormalizado de propósito (copia valores, não referencia) para que o registro do ano nunca mude.'
}
